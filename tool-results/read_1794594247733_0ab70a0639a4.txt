     1→// ============================================================================
     2→// Orders, payments, invoices, subscriptions and fulfillment.
     3→//
     4→// Flow (all prices server-side, immutable snapshots, audited):
     5→//   cart → checkout() → Order(pending_payment)
     6→//   → initiatePayment(orderId, gateway)  [ZarinPal Sandbox | Zibal Sandbox]
     7→//     • server selects + validates the provider (Mellat always rejected)
     8→//     • provider request happens server-side; authority/trackId is stored on
     9→//       the transaction (unique) and the customer's browser is redirected to
    10→//       the REAL provider sandbox payment page
    11→//   → provider redirects the browser to /api/payments/callback/<gateway>
    12→//   → handleGatewayCallback(): server-to-server verification with the
    13→//     provider (the ONLY trust anchor) + amount/order/reference validation
    14→//   → finalizeSuccess() — the single success path: Transaction(succeeded) +
    15→//     Order(paid) + Invoice + Subscription + Service records + Drive quota +
    16→//     notifications + audit.
    17→//   → browser lands on /portal/payment/result?ref=... (real path) which only
    18→//     ever reports server-verified state and re-verifies pending transactions.
    19→//
    20→// The internal mock gateway (dev-only, ABRIT_PAYMENTS_MOCK_ENABLED) keeps the
    21→// legacy token-based finalizePayment path; it is never selectable in
    22→// production. Bank Mellat is disabled and rejected everywhere.
    23→// ============================================================================

    59→function randomRef(prefix: string): string {
    60→  return `${prefix}-${randomInt(10000000, 99999999)}`;
    61→}
    62→
    63→function sha256(value: string): string {
    64→  return createHash("sha256").update(value, "utf8").digest("hex");
    65→}

   178→export const orderService = {
   179→  /** Creates an order from the cart. Prices are recomputed from the inputs. */
   180→  async checkout(
   181→    user: PanelUser,
   182→    input: { invoiceTitle: string; invoiceCompany?: string },
   183→  ): Promise<OrderDTO> {
   184→    const invoiceTitle = String(input.invoiceTitle ?? "").trim();
   185→    if (invoiceTitle.length < 2) {
   186→      throw new HttpError(400, "INVALID_TITLE", "عنوان گیرنده فاکتور را وارد کنید (حداقل ۲ نویسه).");
   187→    }
   188→    const cart = await db.cartItem.findMany({ where: { userId: user.id }, orderBy: { createdAt: "asc" } });
   189→    if (cart.length === 0) {
   190→      throw new HttpError(400, "EMPTY_CART", "سبد خرید شما خالی است.");
   191→    }
   192→
   193→    const items: OrderItemRow[] = [];
   194→    let totalToman = 0;
   195→    for (const row of cart) {
   196→      const quote = computeQuote({
   197→        packageCode: row.packageCode,
   198→        months: row.months,
   199→        users: row.users,
   200→        endpoints: row.endpoints,
   201→      });
   202→      if (!quote.valid || !quote.breakdown || !quote.snapshot) {
   203→        throw new HttpError(409, "ITEM_NOT_QUOTABLE", "یکی از اقلام سبد خرید دیگر قابل قیمت‌گذاری نیست؛ آن را حذف و دوباره انتخاب کنید.");
   204→      }
   205→      if (!verifySnapshotSnapshotChecksum(quote.snapshot)) {
   206→        throw new HttpError(500, "CHECKSUM_FAILED", "خطای یکپارچگی قیمت؛ لطفاً دوباره تلاش کنید.");
   207→      }
   208→      items.push({
   209→        input: {
   210→          packageCode: row.packageCode,
   211→          months: row.months,
   212→          users: row.users,
   213→          endpoints: row.endpoints,
   214→        },
   215→        packageNameFa: quote.breakdown.packageNameFa,
   216→        totalToman: quote.breakdown.totalToman,
   217→        lineItems: quote.lineItems,
   218→        breakdown: quote.breakdown,
   219→        snapshot: quote.snapshot,
   220→      });
   221→      totalToman += quote.breakdown.totalToman;
   222→    }
   223→
   224→    const order = await db.order.create({
   225→      data: {
   226→        orderNumber: randomRef("ABR"),
   227→        userId: user.id,
   228→        status: "pending_payment",
   229→        itemsJson: JSON.stringify(items),
   230→        itemCount: items.length,
   231→        totalToman,
   232→        invoiceTitle,
   233→        invoiceCompany: input.invoiceCompany?.trim() || null,
   234→      },
   235→    });
   236→    await db.cartItem.deleteMany({ where: { userId: user.id } });
   237→
   238→    await audit({
   239→      actor: user,
   240→      action: "order.created",
   241→      entityType: "order",
   242→      entityId: order.id,
   243→      meta: { orderNumber: order.orderNumber, totalToman },
   244→    });
   245→    await notificationService.notify(user.id, {
   246→      category: "order",
   247→      title: "سفارش ثبت شد",
   248→      body: `سفارش ${order.orderNumber} با مبلغ ${totalToman.toLocaleString("en-US")} تومان ثبت شد و در انتظار پرداخت است.`,
   249→      link: "/portal/orders",
   250→    });
   251→
   252→    return toOrderDTO(order);
   253→  },
   254→
   255→  /**
   256→   * Creates a pending transaction with a REAL payment provider (ZarinPal
   257→   * Sandbox / Zibal Sandbox) and returns the provider payment-page URL for a
   258→   * full-page browser redirect. Mellat is rejected here unconditionally.
   259→   */
   260→  async initiatePayment(
   261→    user: PanelUser,
   262→    orderId: string,
   263→    gatewayId: string,
   264→    origin: string,
   265→  ): Promise<PaymentInitDTO> {
   266→    const order = await db.order.findFirst({ where: { id: orderId, userId: user.id } });
   267→    if (!order) throw new HttpError(404, "ORDER_NOT_FOUND", "سفارش یافت نشد.");
   268→    if (order.status !== "pending_payment") {
   269→      throw new HttpError(409, "ORDER_NOT_PAYABLE", "این سفارش در وضعیت قابل پرداخت نیست.");
   270→    }
   271→
   272→    const selection = selectGateway(gatewayId);
   273→    if ("error" in selection) {
   274→      throw new HttpError(400, "GATEWAY_UNAVAILABLE", selection.error);
   275→    }
   276→    const provider = selection.provider;
   277→
   278→    if (provider.id === "mock") {
   279→      // Dev-only internal test gateway (legacy token flow).
   280→      const ref = randomRef("MOCK");
   281→      const verifyToken = randomBytes(24).toString("base64url");
   282→      await db.transaction.create({
   283→        data: {
   284→          ref,
   285→          orderId: order.id,
   286→          userId: user.id,
   287→          gateway: "mock",
   288→          amountToman: order.totalToman,
   289→          status: "pending",
   290→          verifyTokenHash: sha256(verifyToken),
   291→        },
   292→      });
   293→      await audit({
   294→        actor: user,
   295→        action: "payment.initiated",
   296→        entityType: "transaction",
   297→        entityId: ref,
   298→        meta: { gateway: "mock", orderNumber: order.orderNumber, amountToman: order.totalToman },
   299→      });
   300→      return {
   301→        ref,
   302→        gatewayUrl: `/api/payments/gateway/${ref}?t=${encodeURIComponent(verifyToken)}`,
   303→        amountToman: order.totalToman,
   304→      };
   305→    }
   306→
   307→    // Real provider: request the payment server-side.
   308→    const ref = randomRef(provider.id === "zarinpal" ? "ZPL" : "ZBL");
   309→    const callbackUrl = `${origin}/api/payments/callback/${provider.id}`;
   310→    let initiated;
   311→    try {
   312→      initiated = await provider.initiate({
   313→        amountToman: order.totalToman,
   314→        orderNumber: order.orderNumber,
   315→        description: `پرداخت سفارش ${order.orderNumber} — AbrIT`,
   316→        callbackUrl,
   317→      });
   318→    } catch (error) {
   319→      // Provider request failed (network/credentials/rate limit): record the
   320→      // failed attempt for observability and surface a safe error.
   321→      await db.transaction.create({
   322→        data: {
   323→          ref,
   324→          orderId: order.id,
   325→          userId: user.id,
   326→          gateway: provider.id,
   327→          amountToman: order.totalToman,
   328→          status: "failed",
   329→          finalizedAt: new Date(),
   330→          rawResultJson: JSON.stringify({
   331→            phase: "initiate",
   332→            error: error instanceof Error ? error.message : "unknown",
   333→            at: new Date().toISOString(),
   334→          }),
   335→        },
   336→      });
   337→      await audit({
   338→        actor: user,
   339→        action: "payment.initiation_failed",
   340→        entityType: "transaction",
   341→        entityId: ref,
   342→        meta: { gateway: provider.id, orderNumber: order.orderNumber },
   343→      });
   344→      throw new HttpError(
   345→        502,
   346→        "GATEWAY_REQUEST_FAILED",
   347→        "درخواست پرداخت به درگاه انتخابی ارسال نشد؛ لطفاً درگاه دیگری انتخاب کنید یا کمی بعد دوباره تلاش کنید.",
   348→      );
   349→    }
   350→
   351→    await db.transaction.create({
   352→      data: {
   353→        ref,
   354→        orderId: order.id,
   355→        userId: user.id,
   356→        gateway: provider.id,
   357→        amountToman: order.totalToman,
   358→        status: "pending",
   359→        gatewayRef: initiated.gatewayRef,
   360→        rawResultJson: JSON.stringify({ phase: "initiate", at: new Date().toISOString(), result: initiated.raw }),
   361→      },
   362→    });
   363→    await audit({
   364→      actor: user,
   365→      action: "payment.initiated",
   366→      entityType: "transaction",
   367→      entityId: ref,
   368→      meta: { gateway: provider.id, orderNumber: order.orderNumber, amountToman: order.totalToman },
   369→    });
   370→
   371→    return { ref, gatewayUrl: initiated.redirectUrl, amountToman: order.totalToman };
   372→  },
   373→
   374→  /**
   375→   * Handles the provider's browser-redirect callback. Verification happens
   376→   * SERVER-SIDE against the provider API; a browser "success" flag alone is
   377→   * never trusted. Returns the redirect target for the browser (real path).
   378→   */
   379→  async handleGatewayCallback(
   380→    gatewayId: string,
   381→    query: URLSearchParams,
   382→    origin: string,
   383→  ): Promise<string> {
   384→    const selection = selectGateway(gatewayId);
   385→    if ("error" in selection) {
   386→      return `/portal/payment/result?error=${encodeURIComponent(selection.error)}`;
   387→    }
   388→    const provider = selection.provider;
   389→    const resultUrl = (ref: string, extra = "") =>
   390→      `/portal/payment/result?ref=${encodeURIComponent(ref)}${extra}`;
   391→
   392→    // ---- Extract the provider reference from the callback -------------------
   393→    let gatewayRef = "";
   394→    let providerSuccessFlag: boolean | null = null;
   395→    if (provider.id === "zarinpal") {
   396→      gatewayRef = query.get("Authority") ?? query.get("authority") ?? "";
   397→      const status = (query.get("Status") ?? query.get("status") ?? "").toUpperCase();
   398→      providerSuccessFlag = status === "OK" ? true : status === "NOK" ? false : null;
   399→    } else if (provider.id === "zibal") {
   400→      gatewayRef = query.get("trackId") ?? "";
   401→      const success = query.get("success");
   402→      providerSuccessFlag = success === "1" ? true : success === "0" ? false : null;
   403→    }
   404→    if (!gatewayRef) {
   405→      return `/portal/payment/result?error=${encodeURIComponent("پاسخ درگاه پرداخت فاقد شناسه تراکنش است.")}`;
   406→    }
   407→
   408→    const tx = await db.transaction.findUnique({
   409→      where: { gatewayRef },
   410→      include: { order: true },
   411→    });
   412→    if (!tx || tx.gateway !== provider.id) {
   413→      return `/portal/payment/result?error=${encodeURIComponent("تراکنش متناظر با پاسخ درگاه یافت نشد.")}`;
   414→    }
   415→
   416→    // Idempotent: already finalized — just show the final state.
   417→    if (tx.status !== "pending") {
   418→      return resultUrl(tx.ref);
   419→    }
   420→
   421→    // Definitive provider-side failure/cancel — no verification needed.
   422→    if (providerSuccessFlag === false) {
   423→      const cancelled = provider.id === "zibal" && query.get("status") === "2";
   424→      await db.transaction.update({
   425→        where: { id: tx.id },
   426→        data: {
   427→          status: cancelled ? "cancelled" : "failed",
   428→          finalizedAt: new Date(),
   429→          rawResultJson: JSON.stringify({
   430→            phase: "callback",
   431→            outcome: cancelled ? "cancelled" : "failed",
   432→            at: new Date().toISOString(),
   433→            params: Object.fromEntries(query.entries()),
   434→          }),
   435→        },
   436→      });
   437→      await audit({
   438→        actorLabel: `درگاه ${provider.id}`,
   439→        action: cancelled ? "payment.cancelled" : "payment.failed",
   440→        entityType: "transaction",
   441→        entityId: tx.ref,
   442→        meta: { orderNumber: tx.order.orderNumber },
   443→      });
   444→      return resultUrl(tx.ref);
   445→    }
   446→
   447→    // ---- Server-to-server verification (the ONLY trust anchor) -------------
   448→    const outcome = await provider.verify({ amountToman: tx.amountToman, gatewayRef });
   449→    if (outcome.ok) {
   450→      await this.finalizeSuccess(tx, {
   451→        gateway: provider.id,
   452→        alreadyVerified: outcome.alreadyVerified,
   453→        raw: outcome.raw,
   454→      });
   455→      return resultUrl(tx.ref);
   456→    }
   457→    if (outcome.reason === "ambiguous") {
   458→      // Keep pending; the result page re-verifies and supports reconciliation.
   459→      await db.transaction.update({
   460→        where: { id: tx.id },
   461→        data: {
   462→          rawResultJson: JSON.stringify({
   463→            phase: "callback-verify",
   464→            ambiguous: true,
   465→            detail: outcome.detailFa,
   466→            at: new Date().toISOString(),
   467→          }),
   468→        },
   469→      });
   470→      await audit({
   471→        actorLabel: `درگاه ${provider.id}`,
   472→        action: "payment.verification_ambiguous",
   473→        entityType: "transaction",
   474→        entityId: tx.ref,
   475→        meta: { orderNumber: tx.order.orderNumber },
   476→      });
   477→      return resultUrl(tx.ref, "&state=verifying");
   478→    }
   479→    // not_paid / amount_mismatch / invalid_response → definitive failure.
   480→    await db.transaction.update({
   481→      where: { id: tx.id },
   482→      data: {
   483→        status: "failed",
   484→        finalizedAt: new Date(),
   485→        rawResultJson: JSON.stringify({
   486→          phase: "callback-verify",
   487→          reason: outcome.reason,
   488→          at: new Date().toISOString(),
   489→        }),
   490→      },
   491→    });
   492→    await audit({
   493→      actorLabel: `درگاه ${provider.id}`,
   494→      action: "payment.failed",
   495→      entityType: "transaction",
   496→      entityId: tx.ref,
   497→      meta: { reason: outcome.reason, orderNumber: tx.order.orderNumber },
   498→    });
   499→    return resultUrl(tx.ref);
   500→  },

   502→  /**
   503→   * Re-runs server-side verification for a pending transaction (used by the
   504→   * result page while polling, and for reconciliation). Returns the fresh
   505→   * status. Never fabricates a success state.
   506→   */
   507→  async reverifyPendingTransaction(ref: string): Promise<"pending" | "succeeded" | "failed" | "cancelled"> {
   508→    const tx = await db.transaction.findUnique({ where: { ref }, include: { order: true } });
   509→    if (!tx) throw new HttpError(404, "TX_NOT_FOUND", "تراکنش یافت نشد.");
   510→    if (tx.status !== "pending") return tx.status as "succeeded" | "failed" | "cancelled";
   511→    const selection = selectGateway(tx.gateway);
   512→    if ("error" in selection || !tx.gatewayRef) {
   513→      // Non-real gateway rows (mock) simply stay pending until their own
   514→      // callback path finalizes them.
   515→      return "pending";
   516→    }
   517→    const provider: PaymentProvider = selection.provider;
   518→    const outcome = await provider.verify({ amountToman: tx.amountToman, gatewayRef: tx.gatewayRef });
   519→    if (outcome.ok) {
   520→      await this.finalizeSuccess(tx, { gateway: provider.id, alreadyVerified: outcome.alreadyVerified, raw: outcome.raw });
   521→      return "succeeded";
   522→    }
   523→    if (outcome.reason === "not_paid" || outcome.reason === "amount_mismatch") {
   524→      // Definitive negative only when the provider explicitly said so during
   525→      // a re-check (amount mismatches stay failed and flagged for review).
   526→      await db.transaction.update({
   527→        where: { id: tx.id },
   528→        data: {
   529→          status: "failed",
   530→          finalizedAt: new Date(),
   531→          rawResultJson: JSON.stringify({ phase: "reverify", reason: outcome.reason, at: new Date().toISOString() }),
   532→        },
   533→      });
   534→      return "failed";
   535→    }
   536→    return "pending"; // ambiguous — remains pending for reconciliation
   537→  },

   539→  /** Server-rendered gateway page data (validates the URL token). */
   540→  async getTransactionForGateway(ref: string, token: string) {
   541→    const tx = await db.transaction.findUnique({ where: { ref }, include: { order: true } });
   542→    if (!tx) return null;
   543→    if (tx.status !== "pending") return { tx, order: tx.order, valid: false as const };
   544→    if (sha256(token) !== tx.verifyTokenHash) return null;
   545→    return { tx, order: tx.order, valid: true as const };
   546→  },

   548→  /**
   549→   * The single SUCCESS path for every payment (real gateways and the dev-only
   550→   * mock alike). Marks the transaction succeeded, the order paid, issues the
   551→   * invoice and runs fulfillment. Idempotent: an already-finalized
   552→   * transaction is a no-op (duplicate/replayed callbacks are safe).
   553→   */
   554→  async finalizeSuccess(
   555→    tx: Transaction & { order: Order },
   556→    meta: { gateway: string; alreadyVerified?: boolean; raw?: unknown },
   557→  ): Promise<void> {
   558→    if (tx.status === "succeeded") return;
   559→    const now = new Date();
   560→    await db.transaction.update({
   561→      where: { id: tx.id },
   562→      data: {
   563→        status: "succeeded",
   564→        finalizedAt: now,
   565→        rawResultJson: JSON.stringify({
   566→          gateway: meta.gateway,
   567→          verified: true,
   568→          alreadyVerified: meta.alreadyVerified ?? false,
   569→          result: meta.raw ?? null,
   570→          at: now.toISOString(),
   571→        }),
   572→      },
   573→    });
   574→    const order = await db.order.update({
   575→      where: { id: tx.orderId },
   576→      data: { status: "paid", paidAt: now },
   577→    });
   578→    await db.invoice.create({
   579→      data: {
   580→        number: randomRef("INV"),
   581→        orderId: order.id,
   582→        userId: order.userId,
   583→        amountToman: order.totalToman,
   584→        status: "paid",
   585→        title: order.invoiceTitle,
   586→        company: order.invoiceCompany,
   587→        paidAt: now,
   588→      },
   589→    });
   590→    await this.fulfill(order);
   591→    await audit({
   592→      actorLabel: `درگاه ${meta.gateway} (تأیید سمت سرور)`,
   593→      action: "payment.succeeded",
   594→      entityType: "transaction",
   595→      entityId: tx.ref,
   596→      meta: { orderNumber: order.orderNumber, amountToman: order.totalToman },
   597→    });
   598→  },

   600→  /**
   601→   * Legacy finalize for the internal dev/test gateway only (token-based).
   602→   * Real gateways go through handleGatewayCallback + provider verification.
   603→   */
   604→  async finalizePayment(
   605→    ref: string,
   606→    verifyToken: string,
   607→    outcome: "success" | "failure" | "cancel",
   608→  ): Promise<{ status: string; redirectUrl: string }> {
   609→    const tx = await db.transaction.findUnique({ where: { ref }, include: { order: true } });
   610→    if (!tx) throw new HttpError(404, "TX_NOT_FOUND", "تراکنش یافت نشد.");
   611→    if (tx.status !== "pending") {
   612→      // Idempotent: already finalized — return the final state.
   613→      return { status: tx.status, redirectUrl: `/portal/payment/result?ref=${ref}` };
   614→    }
   615→    if (!tx.verifyTokenHash || sha256(verifyToken) !== tx.verifyTokenHash) {
   616→      throw new HttpError(403, "INVALID_TOKEN", "پارامترهای پرداخت معتبر نیست.");
   617→    }
   618→
   619→    if (outcome === "success") {
   620→      await this.finalizeSuccess(tx, { gateway: "mock", raw: { result: "SUCCESS" } });
   621→    } else {
   622→      const now = new Date();
   623→      await db.transaction.update({
   624→        where: { id: tx.id },
   625→        data: {
   626→          status: outcome === "cancel" ? "cancelled" : "failed",
   627→          finalizedAt: now,
   628→          rawResultJson: JSON.stringify({ gateway: "mock", result: outcome.toUpperCase(), at: now.toISOString() }),
   629→        },
   630→      });
   631→      await audit({
   632→        actorLabel: "درگاه داخلی تست",
   633→        action: `payment.${outcome}`,
   634→        entityType: "transaction",
   635→        entityId: ref,
   636→      });
   637→    }
   638→
   639→    return { status: outcome === "success" ? "succeeded" : outcome === "cancel" ? "cancelled" : "failed", redirectUrl: `/portal/payment/result?ref=${ref}` };
   640→  },

   642→  /** Post-payment fulfillment: subscriptions, services, Drive quota, notifications. */
   643→  async fulfill(order: Order & { user?: PanelUser }): Promise<void> {
   644→    const user = order.user ?? (await db.panelUser.findUnique({ where: { id: order.userId } }));
   645→    if (!user) return;
   646→    const items = parseItems(order);
   647→    const now = new Date();
   648→
   649→    for (const item of items) {
   650→      const expiresAt = new Date(now);
   651→      expiresAt.setMonth(expiresAt.getMonth() + item.input.months);
   652→      const sub = await db.subscription.create({
   653→        data: {
   654→          userId: order.userId,
   655→          orderId: order.id,
   656→          packageCode: item.input.packageCode,
   657→          months: item.input.months,
   658→          users: item.input.users,
   659→          endpoints: item.input.endpoints,
   660→          status: "active",
   661→          startsAt: now,
   662→          expiresAt,
   663→          priceTotalToman: item.totalToman,
   664→        },
   665→      });
   666→      await db.service.create({
   667→        data: {
   668→          userId: order.userId,
   669→          subscriptionId: sub.id,
   670→          type: "managed_it",
   671→          title: `خدمات مدیریت‌شده — پکیج ${item.packageNameFa}`,
   672→          status: "active",
   673→          activatedAt: now,
   674→          expiresAt,
   675→          metadataJson: JSON.stringify({
   676→            packageCode: item.input.packageCode,
   677→            packageNameFa: item.packageNameFa,
   678→            users: item.input.users,
   679→            endpoints: item.input.endpoints,
   680→          }),
   681→        },
   682→      });
   683→    }
   684→
   685→    // Invoice notification
   686→    const invoice = await db.invoice.findFirst({ where: { orderId: order.id } });
   687→    await notificationService.notify(user.id, {
   688→      category: "payment",
   689→      title: "پرداخت با موفقیت تأیید شد",
   690→      body: `پرداخت سفارش ${order.orderNumber} تأیید شد${invoice ? ` و فاکتور ${invoice.number} صادر گردید` : ""}. اشتراک شما فعال شد.`,
   691→      link: "/portal/billing",
   692→    });
   693→
   694→    // Apply the entitled Drive quota immediately when a Drive account exists.
   695→    const entitledPlanCode = await getEntitledPlanCode(user.id);
   696→    if (entitledPlanCode) {
   697→      const account = await db.driveAccount.findUnique({ where: { panelUserId: user.id } });
   698→      if (account && account.planCode !== entitledPlanCode) {
   699→        try {
   700→          await driveService.changeQuota(user, entitledPlanCode);
   701→          await notificationService.notify(user.id, {
   702→            category: "drive",
   703→            title: "سهمیه AbrIT Drive به‌روزرسانی شد",
   704→            body: "سهمیه فضای ابری شما مطابق پکیج خریداری‌شده در سرور Drive اعمال شد.",
   705→            link: "/portal/drive",
   706→          });
   707→        } catch (error) {
   708→          console.error("[orders] drive quota application failed:", error);
   709→          await notificationService.notify(user.id, {
   710→            category: "drive",
   711→            title: "اعمال سهمیه Drive ناموفق بود",
   712→            body: "اشتراک شما فعال شد اما اعمال سهمیه جدید روی سرور Drive انجام نشد؛ از صفحه AbrIT Drive همگام‌سازی کنید یا با پشتیبانی تماس بگیرید.",
   713→            link: "/portal/drive",
   714→          });
   715→        }
   716→      } else if (!account) {
   717→        await notificationService.notify(user.id, {
   718→          category: "drive",
   719→          title: "فضای ابری شما آماده فعال‌سازی است",
   720→          body: "با توجه به پکیج خریداری‌شده، می‌توانید حساب AbrIT Drive خود را با سهمیه سازمانی فعال کنید.",
   721→          link: "/portal/drive",
   722→        });
   723→      }
   724→    }

   837→  /** Aggregate for the portal dashboard (all values are real DB data). */
   838→  async portalDashboard(user: PanelUser): Promise<PortalDashboardDTO> {
   839→    const [services, subs, recentOrders, recentTxs, unread, tickets, drive, entitledPlanCode] = await Promise.all([
   840→      db.service.findMany({ where: { userId: user.id } }),
   841→      getActiveSubscriptions(user.id),
   842→      db.order.findMany({ where: { userId: user.id }, orderBy: { createdAt: "desc" }, take: 5 }),
   843→      db.transaction.findMany({ where: { userId: user.id }, orderBy: { createdAt: "desc" }, take: 5, include: { order: true } }),
   844→      notificationService.unreadCount(user.id),
   845→      db.ticket.findMany({ where: { userId: user.id }, select: { status: true } }),
   846→      driveService.getOverview(user).catch(() => null),
   847→      getEntitledPlanCode(user.id),
   848→    ]);
   849→
   850→    const now = Date.now();
   851→    const activeServices = services.filter(
   852→      (s) => s.status === "active" && (!s.expiresAt || s.expiresAt.getTime() > now),
   853→    ).length;
   854→    const bestSub = subs
   855→      .map((s) => ({ sub: s, pkg: getPackage(s.packageCode) }))
   856→      .filter((x) => x.pkg)
   857→      .sort((a, b) => (b.pkg!.driveQuotaBytes - a.pkg!.driveQuotaBytes))[0];
   858→
   859→    return {
   860→      activeServices,
   861→      totalServices: services.length,
   862→      subscription: subs.length
   863→        ? {
   864→            active: true,
   865→            packageNameFa: bestSub?.pkg?.nameFa ?? getPackage(subs[0].packageCode)?.nameFa ?? subs[0].packageCode,
   866→            expiresAt: (bestSub?.sub ?? subs[0]).expiresAt.toISOString(),
   867→          }
   868→        : null,
   869→      entitledPlanCode,
   870→      recentOrders: recentOrders.map(toOrderListEntry),
   871→      recentTransactions: recentTxs.map(toTransactionDTO),
   872→      drive: {
   873→        state: drive?.state ?? "not_provisioned",
   874→        usedBytes: drive?.usage?.usedBytes ?? null,
   875→        totalBytes: drive?.usage?.totalBytes ?? null,
   876→        planNameFa: drive?.account?.planNameFa ?? null,
   877→      },
   878→      unreadNotifications: unread,
   879→      tickets: {
   880→        open: tickets.filter((t) => t.status === "open").length,
   881→        answered: tickets.filter((t) => t.status === "answered").length,
   882→        total: tickets.length,
   883→      },
   884→    };
   885→  },
