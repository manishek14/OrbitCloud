     1→     1→// ============================================================================
     2→     2→// Orders, payments, invoices, subscriptions and fulfillment.
     3→     3→//
     4→     4→// Flow (all prices server-side, immutable snapshots, audited):
     5→     5→//   cart → checkout() → Order(pending_payment)
     6→     6→//   → initiatePayment(orderId, gateway)  [ZarinPal Sandbox | Zibal Sandbox]
     7→     7→//     • server selects + validates the provider (Mellat always rejected)
     8→     8→//     • provider request happens server-side; authority/trackId is stored on
     9→     9→//       the transaction (unique) and the customer's browser is redirected to
    10→    10→//       the REAL provider sandbox payment page
    11→    11→//   → provider redirects the browser to /api/payments/callback/<gateway>
    12→    12→//   → handleGatewayCallback(): server-to-server verification with the
    13→    13→//     provider (the ONLY trust anchor) + amount/order/reference validation
    14→    14→//   → finalizeSuccess() — the single success path: Transaction(succeeded) +
    15→    15→//     Order(paid) + Invoice + Subscription + Service records + Drive quota +
    16→    16→//     notifications + audit.
    17→    17→//   → browser lands on /portal/payment/result?ref=... (real path) which only
    18→    18→//     ever reports server-verified state and re-verifies pending transactions.
    19→    19→//
    20→    20→// The internal mock gateway (dev-only, ABRIT_PAYMENTS_MOCK_ENABLED) keeps the
    21→    21→// legacy token-based finalizePayment path; it is never selectable in
    22→    22→// production. Bank Mellat is disabled and rejected everywhere.

   178→   178→export const orderService = {
   179→   179→  /** Creates an order from the cart. Prices are recomputed from the inputs. */
   180→   180→  async checkout(
   181→   181→    user: PanelUser,
   182→   182→    input: { invoiceTitle: string; invoiceCompany?: string },
   183→   183→  ): Promise<OrderDTO> {
   184→   184→    const invoiceTitle = String(input.invoiceTitle ?? "").trim();
   185→   185→    if (invoiceTitle.length < 2) {
   186→   186→      throw new HttpError(400, "INVALID_TITLE", "عنوان گیرنده فاکتور را وارد کنید (حداقل ۲ نویسه).");
   187→   187→    }
   188→   188→    const cart = await db.cartItem.findMany({ where: { userId: user.id }, orderBy: { createdAt: "asc" } });
   189→   189→    if (cart.length === 0) {
   190→   190→      throw new HttpError(400, "EMPTY_CART", "سبد خرید شما خالی است.");
   191→   191→    }
   192→   192→
   193→   193→    const items: OrderItemRow[] = [];
   194→   194→    let totalToman = 0;
   195→   195→    for (const row of cart) {
   196→   196→      const quote = computeQuote({
   197→   197→        packageCode: row.packageCode,
   198→   198→        months: row.months,
   199→   199→        users: row.users,
   200→   200→        endpoints: row.endpoints,
   201→   201→      });
   202→   202→      if (!quote.valid || !quote.breakdown || !quote.snapshot) {
   203→   203→        throw new HttpError(409, "ITEM_NOT_QUOTABLE", "یکی از اقلام سبد خرید دیگر قابل قیمت‌گذاری نیست؛ آن را حذف و دوباره انتخاب کنید.");
   204→   204→      }
   205→   205→      if (!verifySnapshotSnapshotChecksum(quote.snapshot)) {
   206→   206→        throw new HttpError(500, "CHECKSUM_FAILED", "خطای یکپارچگی قیمت؛ لطفاً دوباره تلاش کنید.");
   207→   207→      }
   208→   208→      items.push({
   209→   209→        input: {
   210→   210→          packageCode: row.packageCode,
   211→   211→          months: row.months,
   212→   212→          users: row.users,
   213→   213→          endpoints: row.endpoints,
   214→   214→        },
   215→   215→        packageNameFa: quote.breakdown.packageNameFa,
   216→   216→        totalToman: quote.breakdown.totalToman,
   217→   217→        lineItems: quote.lineItems,
   218→   218→        breakdown: quote.breakdown,
   219→   219→        snapshot: quote.snapshot,
   220→   220→      });
   221→   221→      totalToman += quote.breakdown.totalToman;
   222→   222→    }
   223→   223→
   224→   224→    const order = await db.order.create({
   225→   225→      data: {
   226→   226→        orderNumber: randomRef("ABR"),
   227→   227→        userId: user.id,
   228→   228→        status: "pending_payment",
   229→   229→        itemsJson: JSON.stringify(items),
   230→   230→        itemCount: items.length,
   231→   231→        totalToman,
   232→   232→        invoiceTitle,
   233→   233→        invoiceCompany: input.invoiceCompany?.trim() || null,
   234→   234→      },
   235→   235→    });
   236→   236→    await db.cartItem.deleteMany({ where: { userId: user.id } });
   237→   237→
   238→   238→    await audit({
   239→   239→      actor: user,
   240→   240→      action: "order.created",
   241→   241→      entityType: "order",
   242→   242→      entityId: order.id,
   243→   243→      meta: { orderNumber: order.orderNumber, totalToman },
   244→   244→    });
   245→   245→    await notificationService.notify(user.id, {
   246→   246→      category: "order",
   247→   247→      title: "سفارش ثبت شد",
   248→   248→      body: `سفارش ${order.orderNumber} با مبلغ ${totalToman.toLocaleString("en-US")} تومان ثبت شد و در انتظار پرداخت است.`,
   249→   249→      link: "/portal/orders",
   250→   250→    });
   251→   251→
   252→   252→    return toOrderDTO(order);
   253→   253→  },
   254→   254→
   255→   255→  /**
   256→   256→   * Creates a pending transaction with a REAL payment provider (ZarinPal
   257→   257→   * Sandbox / Zibal Sandbox) and returns the provider payment-page URL for a
   258→   258→   * full-page browser redirect. Mellat is rejected here unconditionally.
   259→   259→   */
   260→   260→  async initiatePayment(
   261→   261→    user: PanelUser,
   262→   262→    orderId: string,
   263→   263→    gatewayId: string,
   264→   264→    origin: string,
   265→   265→  ): Promise<PaymentInitDTO> {
   266→   266→    const order = await db.order.findFirst({ where: { id: orderId, userId: user.id } });
   267→   267→    if (!order) throw new HttpError(404, "ORDER_NOT_FOUND", "سفارش یافت نشد.");
   268→   268→    if (order.status !== "pending_payment") {
   269→   269→      throw new HttpError(409, "ORDER_NOT_PAYABLE", "این سفارش در وضعیت قابل پرداخت نیست.");
   270→   270→    }
   271→   271→
   272→   272→    const selection = selectGateway(gatewayId);
   273→   273→    if ("error" in selection) {
   274→   274→      throw new HttpError(400, "GATEWAY_UNAVAILABLE", selection.error);
   275→   275→    }
   276→   276→    const provider = selection.provider;
   277→   277→
   278→   278→    if (provider.id === "mock") {
   279→   279→      // Dev-only internal test gateway (legacy token flow).
   280→   280→      const ref = randomRef("MOCK");
   281→   281→      const verifyToken = randomBytes(24).toString("base64url");
   282→   282→      await db.transaction.create({
   283→   283→        data: {
   284→   284→          ref,
   285→   285→          orderId: order.id,
   286→   286→          userId: user.id,
   287→   287→          gateway: "mock",
   288→   288→          amountToman: order.totalToman,
   289→   289→          status: "pending",
   290→   290→          verifyTokenHash: sha256(verifyToken),
   291→   291→        },
   292→   292→      });
   293→   293→      await audit({
   294→   294→        actor: user,
   295→   295→        action: "payment.initiated",
   296→   296→        entityType: "transaction",
   297→   297→        entityId: ref,
   298→   298→        meta: { gateway: "mock", orderNumber: order.orderNumber, amountToman: order.totalToman },
   299→   299→      });
   300→   300→      return {
   301→   301→        ref,
   302→   302→        gatewayUrl: `/api/payments/gateway/${ref}?t=${encodeURIComponent(verifyToken)}`,
   303→   303→        amountToman: order.totalToman,
   304→   304→      };
   305→   305→    }
   306→   306→
   307→   307→    // Real provider: request the payment server-side.
   308→   308→    const ref = randomRef(provider.id === "zarinpal" ? "ZPL" : "ZBL");
   309→   309→    const callbackUrl = `${origin}/api/payments/callback/${provider.id}`;
   310→   310→    let initiated;
   311→   311→    try {
   312→   312→      initiated = await provider.initiate({
   313→   313→        amountToman: order.totalToman,
   314→   314→        orderNumber: order.orderNumber,
   315→   315→        description: `پرداخت سفارش ${order.orderNumber} — AbrIT`,
   316→   316→        callbackUrl,
   317→   317→      });
   318→   318→    } catch (error) {
   319→   319→      // Provider request failed (network/credentials/rate limit): record the
   320→   320→      // failed attempt for observability and surface a safe error.
   321→   321→      await db.transaction.create({
   322→   322→        data: {
   323→   323→          ref,
   324→   324→          orderId: order.id,
   325→   325→          userId: user.id,
   326→   326→          gateway: provider.id,
   327→   327→          amountToman: order.totalToman,
   328→   328→          status: "failed",
   329→   329→          finalizedAt: new Date(),
   330→   330→          rawResultJson: JSON.stringify({
   331→   331→            phase: "initiate",
   332→   332→            error: error instanceof Error ? error.message : "unknown",
   333→   333→            at: new Date().toISOString(),
   334→   334→          }),
   335→   335→        },
   336→   336→      });
   337→   337→      await audit({
   338→   338→        actor: user,
   339→   339→        action: "payment.initiation_failed",
   340→   340→        entityType: "transaction",
   341→   341→        entityId: ref,
   342→   342→        meta: { gateway: provider.id, orderNumber: order.orderNumber },
   343→   343→      });
   344→   344→      throw new HttpError(
   345→   345→        502,
   346→   346→        "GATEWAY_REQUEST_FAILED",
   347→   347→        "درخواست پرداخت به درگاه انتخابی ارسال نشد؛ لطفاً درگاه دیگری انتخاب کنید یا کمی بعد دوباره تلاش کنید.",
   348→   348→      );
   349→   349→    }
   350→   350→
   351→   351→    await db.transaction.create({
   352→   352→      data: {
   353→   353→        ref,
   354→   354→        orderId: order.id,
   355→   355→        userId: user.id,
   356→   356→        gateway: provider.id,
   357→   357→        amountToman: order.totalToman,
   358→   358→        status: "pending",
   359→   359→        gatewayRef: initiated.gatewayRef,
   360→   360→        rawResultJson: JSON.stringify({ phase: "initiate", at: new Date().toISOString(), result: initiated.raw }),
   361→   361→      },
   362→   362→    });
   363→   363→    await audit({
   364→   364→      actor: user,
   365→   365→      action: "payment.initiated",
   366→   366→      entityType: "transaction",
   367→   367→      entityId: ref,
   368→   368→      meta: { gateway: provider.id, orderNumber: order.orderNumber, amountToman: order.totalToman },
   369→   369→    });
   370→   370→
   371→   371→    return { ref, gatewayUrl: initiated.redirectUrl, amountToman: order.totalToman };
   372→   372→  },
   373→   373→
   374→   374→  /**
   375→   375→   * Handles the provider's browser-redirect callback. Verification happens
   376→   376→   * SERVER-SIDE against the provider API; a browser "success" flag alone is
   377→   377→   * never trusted. Returns the redirect target for the browser (real path).
   378→   378→   */
   379→   379→  async handleGatewayCallback(
   380→   380→    gatewayId: string,
   381→   381→    query: URLSearchParams,
   382→   382→    origin: string,
   383→   383→  ): Promise<string> {
   384→   384→    const selection = selectGateway(gatewayId);
   385→   385→    if ("error" in selection) {
   386→   386→      return `/portal/payment/result?error=${encodeURIComponent(selection.error)}`;
   387→   387→    }
   388→   388→    const provider = selection.provider;
   389→   389→    const resultUrl = (ref: string, extra = "") =>
   390→   390→      `/portal/payment/result?ref=${encodeURIComponent(ref)}${extra}`;
   391→   391→
   392→   392→    // ---- Extract the provider reference from the callback -------------------
   393→   393→    let gatewayRef = "";
   394→   394→    let providerSuccessFlag: boolean | null = null;
   395→   395→    if (provider.id === "zarinpal") {
   396→   396→      gatewayRef = query.get("Authority") ?? query.get("authority") ?? "";
   397→   397→      const status = (query.get("Status") ?? query.get("status") ?? "").toUpperCase();
   398→   398→      providerSuccessFlag = status === "OK" ? true : status === "NOK" ? false : null;
   399→   399→    } else if (provider.id === "zibal") {
   400→   400→      gatewayRef = query.get("trackId") ?? "";
   401→   401→      const success = query.get("success");
   402→   402→      providerSuccessFlag = success === "1" ? true : success === "0" ? false : null;
   403→   403→    }
   404→   404→    if (!gatewayRef) {
   405→   405→      return `/portal/payment/result?error=${encodeURIComponent("پاسخ درگاه پرداخت فاقد شناسه تراکنش است.")}`;
   406→   406→    }
   407→   407→
   408→   408→    const tx = await db.transaction.findUnique({
   409→   409→      where: { gatewayRef },
   410→   410→      include: { order: true },
   411→   411→    });
   412→   412→    if (!tx || tx.gateway !== provider.id) {
   413→   413→      return `/portal/payment/result?error=${encodeURIComponent("تراکنش متناظر با پاسخ درگاه یافت نشد.")}`;
   414→   414→    }
   415→   415→
   416→   416→    // Idempotent: already finalized — just show the final state.
   417→   417→    if (tx.status !== "pending") {
   418→   418→      return resultUrl(tx.ref);
   419→   419→    }
   420→   420→
   421→   421→    // Definitive provider-side failure/cancel — no verification needed.
   422→   422→    if (providerSuccessFlag === false) {
   423→   423→      const cancelled = provider.id === "zibal" && query.get("status") === "2";
   424→   424→      await db.transaction.update({
   425→   425→        where: { id: tx.id },
   426→   426→        data: {
   427→   427→          status: cancelled ? "cancelled" : "failed",
   428→   428→          finalizedAt: new Date(),
   429→   429→          rawResultJson: JSON.stringify({
   430→   430→            phase: "callback",
   431→   431→            outcome: cancelled ? "cancelled" : "failed",
   432→   432→            at: new Date().toISOString(),
   433→   433→            params: Object.fromEntries(query.entries()),
   434→   434→          }),
   435→   435→        },
   436→   436→      });
   437→   437→      await audit({
   438→   438→        actorLabel: `درگاه ${provider.id}`,
   439→   439→        action: cancelled ? "payment.cancelled" : "payment.failed",
   440→   440→        entityType: "transaction",
   441→   441→        entityId: tx.ref,
   442→   442→        meta: { orderNumber: tx.order.orderNumber },
   443→   443→      });
   444→   444→      return resultUrl(tx.ref);
   445→   445→    }
   446→   446→
   447→   447→    // ---- Server-to-server verification (the ONLY trust anchor) -------------
   448→   448→    const outcome = await provider.verify({ amountToman: tx.amountToman, gatewayRef });
   449→   449→    if (outcome.ok) {
   450→   450→      await this.finalizeSuccess(tx, {
   451→   451→        gateway: provider.id,
   452→   452→        alreadyVerified: outcome.alreadyVerified,
   453→   453→        raw: outcome.raw,
   454→   454→      });
   455→   455→      return resultUrl(tx.ref);
   456→   456→    }
   457→   457→    if (outcome.reason === "ambiguous") {
   458→   458→      // Keep pending; the result page re-verifies and supports reconciliation.
   459→   459→      await db.transaction.update({
   460→   460→        where: { id: tx.id },
   461→   461→        data: {
   462→   462→          rawResultJson: JSON.stringify({
   463→   463→            phase: "callback-verify",
   464→   464→            ambiguous: true,
   465→   465→            detail: outcome.detailFa,
   466→   466→            at: new Date().toISOString(),
   467→   467→          }),
   468→   468→        },
   469→   469→      });
   470→   470→      await audit({
   471→   471→        actorLabel: `درگاه ${provider.id}`,
   472→   472→        action: "payment.verification_ambiguous",
   473→   473→        entityType: "transaction",
   474→   474→        entityId: tx.ref,
   475→   475→        meta: { orderNumber: tx.order.orderNumber },
   476→   476→      });
   477→   477→      return resultUrl(tx.ref, "&state=verifying");
   478→   478→    }
   479→   479→    // not_paid / amount_mismatch / invalid_response → definitive failure.
   480→   480→    await db.transaction.update({
   481→   481→      where: { id: tx.id },
   482→   482→      data: {
   483→   483→        status: "failed",
   484→   484→        finalizedAt: new Date(),
   485→   485→        rawResultJson: JSON.stringify({
   486→   486→          phase: "callback-verify",
   487→   487→          reason: outcome.reason,
   488→   488→          at: new Date().toISOString(),
   489→   489→        }),
   490→   490→      },
   491→   491→    });
   492→   492→    await audit({
   493→   493→      actorLabel: `درگاه ${provider.id}`,
   494→   494→      action: "payment.failed",
   495→   495→      entityType: "transaction",
   496→   496→      entityId: tx.ref,
   497→   497→      meta: { reason: outcome.reason, orderNumber: tx.order.orderNumber },
   498→   498→    });
   499→   499→    return resultUrl(tx.ref);
   500→   500→  },
   501→   501→
   502→   502→  /**
   503→   503→   * Re-runs server-side verification for a pending transaction (used by the
   504→   504→   * result page while polling, and for reconciliation). Returns the fresh
   505→   505→   * status. Never fabricates a success state.
   506→   506→   */
   507→   507→  async reverifyPendingTransaction(ref: string): Promise<"pending" | "succeeded" | "failed" | "cancelled"> {
   508→   508→    const tx = await db.transaction.findUnique({ where: { ref }, include: { order: true } });
   509→   509→    if (!tx) throw new HttpError(404, "TX_NOT_FOUND", "تراکنش یافت نشد.");
   510→   510→    if (tx.status !== "pending") return tx.status as "succeeded" | "failed" | "cancelled";
   511→   511→    const selection = selectGateway(tx.gateway);
   512→   512→    if ("error" in selection || !tx.gatewayRef) {
   513→   513→      // Non-real gateway rows (mock) simply stay pending until their own
   514→   514→      // callback path finalizes them.
   515→   515→      return "pending";
   516→   516→    }
   517→   517→    const provider: PaymentProvider = selection.provider;
   518→   518→    const outcome = await provider.verify({ amountToman: tx.amountToman, gatewayRef: tx.gatewayRef });
   519→   519→    if (outcome.ok) {
   520→   520→      await this.finalizeSuccess(tx, { gateway: provider.id, alreadyVerified: outcome.alreadyVerified, raw: outcome.raw });
   521→   521→      return "succeeded";
   522→   522→    }
   523→   523→    if (outcome.reason === "not_paid" || outcome.reason === "amount_mismatch") {
   524→   524→      // Definitive negative only when the provider explicitly said so during
   525→   525→      // a re-check (amount mismatches stay failed and flagged for review).
   526→   526→      await db.transaction.update({
   527→   527→        where: { id: tx.id },
   528→   528→        data: {
   529→   529→          status: "failed",
   530→   530→          finalizedAt: new Date(),
   531→   531→          rawResultJson: JSON.stringify({ phase: "reverify", reason: outcome.reason, at: new Date().toISOString() }),
   532→   532→        },
   533→   533→      });
   534→   534→      return "failed";
   535→   535→    }
   536→   536→    return "pending"; // ambiguous — remains pending for reconciliation
   537→   537→  },

   539→   539→  /** Server-rendered gateway page data (validates the URL token). */
   540→   540→  async getTransactionForGateway(ref: string, token: string) {
   541→   541→    const tx = await db.transaction.findUnique({ where: { ref }, include: { order: true } });
   542→   542→    if (!tx) return null;
   543→   543→    if (tx.status !== "pending") return { tx, order: tx.order, valid: false as const };
   544→   544→    if (sha256(token) !== tx.verifyTokenHash) return null;
   545→   545→    return { tx, order: tx.order, valid: true as const };
   546→   546→  },

   548→   548→  /**
   549→   549→   * The single SUCCESS path for every payment (real gateways and the dev-only
   550→   550→   * mock alike). Marks the transaction succeeded, the order paid, issues the
   551→   551→   * invoice and runs fulfillment. Idempotent: an already-finalized
   552→   552→   * transaction is a no-op (duplicate/replayed callbacks are safe).
   553→   553→   */
   554→   554→  async finalizeSuccess(
   555→   555→    tx: Transaction & { order: Order },
   556→   556→    meta: { gateway: string; alreadyVerified?: boolean; raw?: unknown },
   557→   557→  ): Promise<void> {
   558→   558→    if (tx.status === "succeeded") return;
   559→   559→    const now = new Date();
   560→   560→    await db.transaction.update({
   561→   561→      where: { id: tx.id },
   562→   562→      data: {
   563→   563→        status: "succeeded",
   564→   564→        finalizedAt: now,
   565→   565→        rawResultJson: JSON.stringify({
   566→   566→          gateway: meta.gateway,
   567→   567→          verified: true,
   568→   568→          alreadyVerified: meta.alreadyVerified ?? false,
   569→   569→          result: meta.raw ?? null,
   570→   570→          at: now.toISOString(),
   571→   571→        }),
   572→   572→      },
   573→   573→    });
   574→   574→    const order = await db.order.update({
   575→   575→      where: { id: tx.orderId },
   576→   576→      data: { status: "paid", paidAt: now },
   577→   577→    });
   578→   578→    await db.invoice.create({
   579→   579→      data: {
   580→   580→        number: randomRef("INV"),
   581→   581→        orderId: order.id,
   582→   582→        userId: order.userId,
   583→   583→        amountToman: order.totalToman,
   584→   584→        status: "paid",
   585→   585→        title: order.invoiceTitle,
   586→   586→        company: order.invoiceCompany,
   587→   587→        paidAt: now,
   588→   588→      },
   589→   589→    });
   590→   590→    await this.fulfill(order);
   591→   591→    await audit({
   592→   592→      actorLabel: `درگاه ${meta.gateway} (تأیید سمت سرور)`,
   593→   593→      action: "payment.succeeded",
   594→   594→      entityType: "transaction",
   595→   595→      entityId: tx.ref,
   596→   596→      meta: { orderNumber: order.orderNumber, amountToman: order.totalToman },
   597→   597→    });
   598→   598→  },

   600→   600→  /**
   601→   601→   * Legacy finalize for the internal dev/test gateway only (token-based).
   602→   602→   * Real gateways go through handleGatewayCallback + provider verification.
   603→   603→   */
   604→   604→  async finalizePayment(
   605→   605→    ref: string,
   606→   606→    verifyToken: string,
   607→   607→    outcome: "success" | "failure" | "cancel",
   608→   608→  ): Promise<{ status: string; redirectUrl: string }> {
   609→   609→    const tx = await db.transaction.findUnique({ where: { ref }, include: { order: true } });
   610→   610→    if (!tx) throw new HttpError(404, "TX_NOT_FOUND", "تراکنش یافت نشد.");
   611→   611→    if (tx.status !== "pending") {
   612→   612→      // Idempotent: already finalized — return the final state.
   613→   613→      return { status: tx.status, redirectUrl: `/portal/payment/result?ref=${ref}` };
   614→   614→    }
   615→   615→    if (!tx.verifyTokenHash || sha256(verifyToken) !== tx.verifyTokenHash) {
   616→   616→      throw new HttpError(403, "INVALID_TOKEN", "پارامترهای پرداخت معتبر نیست.");
   617→   617→    }
   618→   618→
   619→   619→    if (outcome === "success") {
   620→   620→      await this.finalizeSuccess(tx, { gateway: "mock", raw: { result: "SUCCESS" } });
   621→   621→    } else {
   622→   622→      const now = new Date();
   623→   623→      await db.transaction.update({
   624→   624→        where: { id: tx.id },
   625→   625→        data: {
   626→   626→          status: outcome === "cancel" ? "cancelled" : "failed",
   627→   627→          finalizedAt: now,
   628→   628→          rawResultJson: JSON.stringify({ gateway: "mock", result: outcome.toUpperCase(), at: now.toISOString() }),
   629→   629→        },
   630→   630→      });
   631→   631→      await audit({
   632→   632→        actorLabel: "درگاه داخلی تست",
   633→   633→        action: `payment.${outcome}`,
   634→   634→        entityType: "transaction",
   635→   635→        entityId: ref,
   636→   636→      });
   637→   637→    }
   638→   638→
   639→   639→    return { status: outcome === "success" ? "succeeded" : outcome === "cancel" ? "cancelled" : "failed", redirectUrl: `/portal/payment/result?ref=${ref}` };
   640→   640→  },
